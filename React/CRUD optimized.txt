⚙️ Optimized Version
import React, { useState, useEffect, useCallback } from "react";

function App() {
  const [formData, setFormData] = useState({
    name: "",
    age: "",
    email: "",
    phone: "",
    city: "",
  });

  const [data, setData] = useState([]);
  const [isEdit, setIsEdit] = useState(false);
  const [editIndex, setEditIndex] = useState(null);

  const buttonText = isEdit ? "Save" : "Submit";

  // 🔹 Handle Input Change (Single Handler for All Inputs)
  const handleChange = useCallback((e) => {
    const { name, value } = e.target;
    setFormData((prev) => ({ ...prev, [name]: value }));
  }, []);

  // 🔹 Handle Submit / Save
  const handleSubmit = useCallback(
    (e) => {
      e.preventDefault();
      const newEntry = { ...formData, isChecked: false };

      setData((prevData) =>
        isEdit
          ? prevData.map((item, idx) =>
              idx === editIndex ? newEntry : item
            )
          : [...prevData, newEntry]
      );

      // Reset form
      setFormData({ name: "", age: "", email: "", phone: "", city: "" });
      setIsEdit(false);
      setEditIndex(null);
    },
    [formData, isEdit, editIndex]
  );

  // 🔹 Handle Edit
  const handleEdit = useCallback((index) => {
    setFormData(data[index]);
    setIsEdit(true);
    setEditIndex(index);
  }, [data]);

  // 🔹 Handle Checkbox Toggle
  const handleCheckbox = useCallback((index) => {
    setData((prev) =>
      prev.map((item, i) =>
        i === index ? { ...item, isChecked: !item.isChecked } : item
      )
    );
  }, []);

  // 🔹 Handle Delete Selected
  const handleDelete = useCallback(() => {
    setData((prev) => prev.filter((item) => !item.isChecked));
  }, []);

  // 🔹 Handle Clear All
  const handleClearAll = useCallback(() => {
    if (window.confirm("Are you sure you want to clear all data?")) {
      setData([]);
    }
  }, []);

  return (
    <div style={{ padding: "20px", display: "flex", gap: "40px" }}>
      {/* ----------- FORM ----------- */}
      <form onSubmit={handleSubmit}>
        <table>
          <tbody>
            {["name", "age", "email", "phone", "city"].map((field) => (
              <tr key={field}>
                <td>{field.charAt(0).toUpperCase() + field.slice(1)}</td>
                <td>
                  <input
                    type={field === "email" ? "email" : "text"}
                    name={field}
                    value={formData[field]}
                    onChange={handleChange}
                    required
                  />
                </td>
              </tr>
            ))}
            <tr>
              <td></td>
              <td>
                <button type="submit">{buttonText}</button>
              </td>
            </tr>
          </tbody>
        </table>
      </form>

      {/* ----------- TABLE ----------- */}
      <div id="table">
        <table border="1" cellPadding="8">
          <thead>
            <tr>
              <th>Name</th>
              <th>Age</th>
              <th>Email</th>
              <th>Phone</th>
              <th>City</th>
              <th>Edit</th>
              <th>Select</th>
            </tr>
          </thead>
          <tbody>
            {data.map((item, index) => (
              <tr key={index}>
                <td>{item.name}</td>
                <td>{item.age}</td>
                <td>{item.email}</td>
                <td>{item.phone}</td>
                <td>{item.city}</td>
                <td>
                  <button onClick={() => handleEdit(index)}>Edit</button>
                </td>
                <td>
                  <input
                    type="checkbox"
                    checked={item.isChecked}
                    onChange={() => handleCheckbox(index)}
                  />
                </td>
              </tr>
            ))}
          </tbody>
        </table>

        {/* ----------- ACTION BUTTONS ----------- */}
        {data.length > 0 && (
          <div style={{ marginTop: "10px" }}>
            {data.some((item) => item.isChecked) && (
              <button
                onClick={handleDelete}
                style={{
                  backgroundColor: "red",
                  color: "white",
                  marginRight: "10px",
                }}
              >
                Delete Selected
              </button>
            )}
            <button
              onClick={handleClearAll}
              style={{ backgroundColor: "gray", color: "white" }}
            >
              Clear All
            </button>
          </div>
        )}
      </div>
    </div>
  );
}

export default App;

🚀 Key Optimizations

Single Input Handler (handleChange) → replaced 5 separate onChange functions.

Used useCallback → prevents unnecessary re-renders of child functions.

Dynamic Input Rendering → mapped over fields instead of repeating code.

Simplified Button Text Logic → removed extra state and effect.

State Reset Logic Centralized → one clean reset after submit.

Avoided Inline State Mutations → all updates done via updater functions.
